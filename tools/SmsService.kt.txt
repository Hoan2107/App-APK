package __PACKAGE__

import android.app.Notification
import android.app.NotificationChannel
import android.app.NotificationManager
import android.app.PendingIntent
import android.app.Service
import android.content.Context
import android.content.Intent
import android.content.pm.ServiceInfo
import android.os.Build
import android.os.Handler
import android.os.IBinder
import android.os.Looper
import android.os.PowerManager
import android.telephony.SmsManager

/** Trạng thái công việc gửi, dùng chung giữa giao diện và dịch vụ nền (cùng 1 tiến trình). */
object SmsJob {
    @Volatile var running = false
    var names = ArrayList<String>()
    var phones = ArrayList<String>()
    var message = ""

    /** Mỗi người 1 ký tự: P = chờ, S = đã gửi, F = lỗi. */
    var results = StringBuilder()
    var errors = HashMap<Int, String>()

    @Volatile var index = 0
    @Volatile var sent = 0
    @Volatile var failed = 0
    @Volatile var nextAt = 0L
    @Volatile var delayMs = 15000L

    fun clear() {
        running = false
        names = ArrayList<String>()
        phones = ArrayList<String>()
        message = ""
        results = StringBuilder()
        errors = HashMap<Int, String>()
        index = 0
        sent = 0
        failed = 0
        nextAt = 0L
    }
}

/** Dịch vụ chạy nền: cứ mỗi delayMs gửi 1 tin cho 1 người, kể cả khi tắt màn hình. */
class SmsService : Service() {
    companion object {
        const val CHANNEL_ID = "sms_bulk_channel"
        const val NOTIF_ID = 4711
    }

    private val namePattern =
        Regex("\\{+\\s*(ten|tên|name)\\s*\\}+", RegexOption.IGNORE_CASE)
    private val handler = Handler(Looper.getMainLooper())
    private var wakeLock: PowerManager.WakeLock? = null
    private val tick = Runnable { sendNext() }

    override fun onBind(intent: Intent?): IBinder? = null

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        createChannel()
        goForeground(buildNotification("Đang chuẩn bị gửi...", true))
        acquireWakeLock()

        if (SmsJob.index >= SmsJob.phones.size) {
            finishJob()
            return START_NOT_STICKY
        }
        SmsJob.running = true
        handler.removeCallbacks(tick)
        handler.post(tick)
        return START_NOT_STICKY
    }

    override fun onDestroy() {
        handler.removeCallbacks(tick)
        SmsJob.running = false
        SmsJob.nextAt = 0L
        releaseWakeLock()
        super.onDestroy()
    }

    private fun sendNext() {
        if (!SmsJob.running) return
        val i = SmsJob.index
        if (i >= SmsJob.phones.size) {
            finishJob()
            return
        }

        val name = SmsJob.names[i]
        val phone = SmsJob.phones[i]
        val display = if (name.isEmpty()) "bạn" else name
        // Thay {ten}/{Ten}/{TEN}/{{ten}}... bằng tên thật, rồi xoá mọi dấu { } còn sót
        // để tin nhắn gửi đi tuyệt đối không có ngoặc nhọn.
        val text = namePattern.replace(SmsJob.message) { display }
            .replace("{", "")
            .replace("}", "")

        try {
            sendSms(phone, text)
            SmsJob.sent = SmsJob.sent + 1
            SmsJob.results.setCharAt(i, 'S')
        } catch (e: Exception) {
            SmsJob.failed = SmsJob.failed + 1
            SmsJob.errors[i] = e.message ?: e.toString()
            SmsJob.results.setCharAt(i, 'F')
        }
        SmsJob.index = i + 1

        if (SmsJob.index >= SmsJob.phones.size) {
            finishJob()
            return
        }

        val nextName = SmsJob.names[SmsJob.index]
        val nextLabel = if (nextName.isEmpty()) SmsJob.phones[SmsJob.index] else nextName
        updateNotification(
            "Đã gửi ${SmsJob.sent}/${SmsJob.phones.size} · tiếp theo: $nextLabel",
            true
        )
        SmsJob.nextAt = System.currentTimeMillis() + SmsJob.delayMs
        handler.postDelayed(tick, SmsJob.delayMs)
    }

    private fun finishJob() {
        SmsJob.running = false
        SmsJob.nextAt = 0L
        handler.removeCallbacks(tick)
        updateNotification(
            "Hoàn tất: ${SmsJob.sent} thành công, ${SmsJob.failed} lỗi",
            false
        )
        if (Build.VERSION.SDK_INT >= 24) {
            stopForeground(Service.STOP_FOREGROUND_DETACH)
        } else {
            @Suppress("DEPRECATION")
            stopForeground(false)
        }
        stopSelf()
    }

    private fun sendSms(phone: String, message: String) {
        val sms: SmsManager =
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.S) {
                getSystemService(SmsManager::class.java)
            } else {
                @Suppress("DEPRECATION")
                SmsManager.getDefault()
            }
        val parts = sms.divideMessage(message)
        if (parts.size > 1) {
            sms.sendMultipartTextMessage(phone, null, parts, null, null)
        } else {
            sms.sendTextMessage(phone, null, message, null, null)
        }
    }

    private fun acquireWakeLock() {
        try {
            if (wakeLock == null) {
                val pm = getSystemService(Context.POWER_SERVICE) as PowerManager
                wakeLock = pm.newWakeLock(PowerManager.PARTIAL_WAKE_LOCK, "smstool:sending")
                wakeLock?.setReferenceCounted(false)
            }
            if (wakeLock?.isHeld != true) {
                wakeLock?.acquire(24L * 60L * 60L * 1000L)
            }
        } catch (e: Exception) {
            // Bỏ qua: vẫn chạy được, chỉ kém ổn định khi tắt màn hình.
        }
    }

    private fun releaseWakeLock() {
        try {
            if (wakeLock?.isHeld == true) wakeLock?.release()
        } catch (e: Exception) {
        }
        wakeLock = null
    }

    private fun createChannel() {
        if (Build.VERSION.SDK_INT >= 26) {
            val nm = getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager
            val channel = NotificationChannel(
                CHANNEL_ID,
                "Gửi SMS tự động",
                NotificationManager.IMPORTANCE_LOW
            )
            nm.createNotificationChannel(channel)
        }
    }

    private fun buildNotification(text: String, ongoing: Boolean): Notification {
        val builder: Notification.Builder =
            if (Build.VERSION.SDK_INT >= 26) {
                Notification.Builder(this, CHANNEL_ID)
            } else {
                @Suppress("DEPRECATION")
                Notification.Builder(this)
            }
        builder
            .setSmallIcon(android.R.drawable.ic_dialog_email)
            .setContentTitle("SMS Tool")
            .setContentText(text)
            .setOngoing(ongoing)
            .setOnlyAlertOnce(true)

        val launch = packageManager.getLaunchIntentForPackage(packageName)
        if (launch != null) {
            val pi = PendingIntent.getActivity(
                this,
                0,
                launch,
                PendingIntent.FLAG_IMMUTABLE or PendingIntent.FLAG_UPDATE_CURRENT
            )
            builder.setContentIntent(pi)
        }
        return builder.build()
    }

    private fun updateNotification(text: String, ongoing: Boolean) {
        try {
            val nm = getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager
            nm.notify(NOTIF_ID, buildNotification(text, ongoing))
        } catch (e: Exception) {
        }
    }

    private fun goForeground(n: Notification) {
        try {
            if (Build.VERSION.SDK_INT >= 34) {
                startForeground(NOTIF_ID, n, ServiceInfo.FOREGROUND_SERVICE_TYPE_SPECIAL_USE)
            } else {
                startForeground(NOTIF_ID, n)
            }
        } catch (e: Exception) {
            // Nếu hệ thống từ chối, vẫn thử chạy tiếp khi app đang mở.
        }
    }
}
