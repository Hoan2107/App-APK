package __PACKAGE__

import android.Manifest
import android.content.Intent
import android.content.pm.PackageManager
import android.os.Build
import io.flutter.embedding.android.FlutterActivity
import io.flutter.embedding.engine.FlutterEngine
import io.flutter.plugin.common.MethodChannel

class MainActivity : FlutterActivity() {
    private val channelName = "sms_bulk"
    private val permissionRequestCode = 1001
    private var pendingResult: MethodChannel.Result? = null

    private fun hasSmsPermission(): Boolean {
        return if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.M) {
            checkSelfPermission(Manifest.permission.SEND_SMS) == PackageManager.PERMISSION_GRANTED
        } else {
            true
        }
    }

    private fun launchService() {
        val intent = Intent(this, SmsService::class.java)
        if (Build.VERSION.SDK_INT >= 26) {
            startForegroundService(intent)
        } else {
            startService(intent)
        }
    }

    private fun stopSending() {
        try {
            stopService(Intent(this, SmsService::class.java))
        } catch (e: Exception) {
        }
        SmsJob.running = false
        SmsJob.nextAt = 0L
    }

    override fun configureFlutterEngine(flutterEngine: FlutterEngine) {
        super.configureFlutterEngine(flutterEngine)
        MethodChannel(flutterEngine.dartExecutor.binaryMessenger, channelName)
            .setMethodCallHandler { call, result ->
                when (call.method) {
                    "requestPermission" -> {
                        val needed = ArrayList<String>()
                        if (!hasSmsPermission()) {
                            needed.add(Manifest.permission.SEND_SMS)
                        }
                        if (Build.VERSION.SDK_INT >= 33 &&
                            checkSelfPermission(Manifest.permission.POST_NOTIFICATIONS) !=
                            PackageManager.PERMISSION_GRANTED
                        ) {
                            needed.add(Manifest.permission.POST_NOTIFICATIONS)
                        }
                        if (needed.isEmpty() || Build.VERSION.SDK_INT < Build.VERSION_CODES.M) {
                            result.success(hasSmsPermission())
                        } else {
                            pendingResult = result
                            requestPermissions(needed.toTypedArray(), permissionRequestCode)
                        }
                    }

                    "start" -> {
                        val list = call.argument<List<*>>("contacts") ?: emptyList<Any>()
                        val message = call.argument<String>("message") ?: ""
                        val delaySec = call.argument<Int>("delaySec") ?: 15

                        if (SmsJob.running) {
                            result.error("BUSY", "Đang gửi, hãy dừng trước", null)
                        } else {
                            SmsJob.clear()
                            for (item in list) {
                                val m = item as? Map<*, *> ?: continue
                                val phone = m["phone"]?.toString() ?: continue
                                SmsJob.names.add(m["name"]?.toString() ?: "")
                                SmsJob.phones.add(phone)
                            }
                            if (SmsJob.phones.isEmpty() || message.isBlank()) {
                                SmsJob.clear()
                                result.error("BAD_ARGS", "Thiếu danh sách hoặc nội dung", null)
                            } else if (!hasSmsPermission()) {
                                SmsJob.clear()
                                result.error("NO_PERMISSION", "Chưa cấp quyền gửi SMS", null)
                            } else {
                                SmsJob.message = message
                                SmsJob.delayMs = delaySec * 1000L
                                SmsJob.results = StringBuilder("P".repeat(SmsJob.phones.size))
                                SmsJob.times = ArrayList<Long>(List(SmsJob.phones.size) { 0L })
                                SmsJob.running = true
                                try {
                                    launchService()
                                    result.success(true)
                                } catch (e: Exception) {
                                    SmsJob.running = false
                                    result.error("START_FAILED", e.message ?: e.toString(), null)
                                }
                            }
                        }
                    }

                    "resume" -> {
                        val message = call.argument<String>("message")
                        if (SmsJob.running) {
                            result.success(true)
                        } else if (SmsJob.phones.isEmpty() || SmsJob.index >= SmsJob.phones.size) {
                            result.error("NO_JOB", "Không còn người nào để gửi tiếp", null)
                        } else if (!hasSmsPermission()) {
                            result.error("NO_PERMISSION", "Chưa cấp quyền gửi SMS", null)
                        } else {
                            if (!message.isNullOrBlank()) {
                                SmsJob.message = message
                            }
                            SmsJob.running = true
                            try {
                                launchService()
                                result.success(true)
                            } catch (e: Exception) {
                                SmsJob.running = false
                                result.error("START_FAILED", e.message ?: e.toString(), null)
                            }
                        }
                    }

                    "stop" -> {
                        stopSending()
                        result.success(true)
                    }

                    "clear" -> {
                        stopSending()
                        SmsJob.clear()
                        result.success(true)
                    }

                    "status" -> {
                        val errs = HashMap<String, String>()
                        for ((k, v) in SmsJob.errors) {
                            errs[k.toString()] = v
                        }
                        result.success(
                            hashMapOf<String, Any>(
                                "running" to SmsJob.running,
                                "total" to SmsJob.phones.size,
                                "sent" to SmsJob.sent,
                                "failed" to SmsJob.failed,
                                "index" to SmsJob.index,
                                "nextAt" to SmsJob.nextAt,
                                "results" to SmsJob.results.toString(),
                                "times" to ArrayList<Long>(SmsJob.times),
                                "errors" to errs
                            )
                        )
                    }

                    "getJob" -> {
                        result.success(
                            hashMapOf<String, Any>(
                                "total" to SmsJob.phones.size,
                                "names" to ArrayList<String>(SmsJob.names),
                                "phones" to ArrayList<String>(SmsJob.phones),
                                "message" to SmsJob.message
                            )
                        )
                    }

                    else -> result.notImplemented()
                }
            }
    }

    override fun onRequestPermissionsResult(
        requestCode: Int,
        permissions: Array<out String>,
        grantResults: IntArray
    ) {
        super.onRequestPermissionsResult(requestCode, permissions, grantResults)
        if (requestCode == permissionRequestCode) {
            pendingResult?.success(hasSmsPermission())
            pendingResult = null
        }
    }
}
